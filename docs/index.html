Hard ban

Spawn of general-purpose and explore fails (case-insensitive), even if toggled or shadowed.

Task ads

Built-in roster is plan only. Defaults no longer open a GP loop.

Full-tool alias

First-party goal / scheduler / workflow use non-banned agent for full capability.

6h watch

Timer-based re-apply is optional; manual patching is the default.

Install

# clone + gates
git clone https://github.com/VeigaPunk/grok-build-livepatch.git
cd grok-build-livepatch
chmod +x scripts/*.sh
./scripts/gates.sh
./scripts/check-and-patch.sh          # network + cargo (first run)
./scripts/install-timer.sh            # optional: 6h user timer, REPLACE_BIN=1
./scripts/install-timer.sh --link-bin
./scripts/install-timer.sh --status   # active_cli=livepatch

# marketplace stack (agents / skills / commands)
grok plugin marketplace add VeigaPunk/grok-marketplace
grok plugin install xbgst-stack@veigapunk/grok-marketplace --trust
bash ~/.grok/installed-plugins/xbgst-stack-*/scripts/install-host.sh
# add --install-timer only to opt in to scheduling

Recommended CLI config

Prefer the primary site config. This mirror serves the same template as docs/cli-config.toml in the livepatch repo. Merge into ~/.grok/config.toml.

[cli]
installer = "internal"

[marketplace]
default_skills_installs_purged = true
official_marketplace_auto_installed = true

[[marketplace.sources]]
name = "xAI Official"
git = "https://github.com/xai-org/plugin-marketplace.git"

[[marketplace.sources]]
name = "grok-marketplace"
git = "https://github.com/VeigaPunk/grok-marketplace.git"

[[marketplace.sources]]
name = "ds4cc-marketplace"

[ui]
max_thoughts_width = 120
fork_secondary_model = "grok-4.5"
yolo = false
compact_mode = false
permission_mode = "always-approve"

[plugins]

    "tinyfish",
    "cloudflare",
    "xbgst-stack",
    "heuer-planning",
]

[models]
default = "grok-4.5"

[subagents]

max_depth = 1
max_concurrent = 16

[subagents.toggle]
# Defense in depth: livepatch hard-rejects these at spawn.
explore = false
general-purpose = false
